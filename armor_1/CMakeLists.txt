cmake_minimum_required(VERSION 3.16)
project(armor_extra LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(OpenCV REQUIRED COMPONENTS core imgproc videoio highgui calib3d dnn)
add_executable(team_armor team_armor.cpp)
target_include_directories(team_armor PRIVATE ${OpenCV_INCLUDE_DIRS})
target_link_libraries(team_armor PRIVATE ${OpenCV_LIBS})
target_compile_options(team_armor PRIVATE -Wall -Wextra -Wpedantic)
